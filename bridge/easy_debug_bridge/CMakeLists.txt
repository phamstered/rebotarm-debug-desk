cmake_minimum_required(VERSION 3.16)
project(easy_debug_bridge)

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)

option(EASY_DEBUG_OBJECT_DETECTION "Enable PCL tabletop object detection when available" ON)
if(EASY_DEBUG_OBJECT_DETECTION)
  find_package(PCL QUIET COMPONENTS common filters segmentation search sample_consensus)
endif()
add_library(object_detection STATIC src/object_detection.cpp)
target_include_directories(object_detection PUBLIC include PRIVATE ${Boost_INCLUDE_DIRS})
if(PCL_FOUND AND EASY_DEBUG_OBJECT_DETECTION)
  target_compile_definitions(object_detection PRIVATE EASY_DEBUG_HAS_PCL=1)
  target_include_directories(object_detection SYSTEM PRIVATE ${PCL_INCLUDE_DIRS})
  target_link_libraries(object_detection PUBLIC ${PCL_LIBRARIES})
else()
  message(STATUS "PCL tabletop detection disabled; RGBD preview remains available")
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d)
add_library(rgb_target STATIC src/rgb_target.cpp)
target_compile_definitions(rgb_target PRIVATE EASY_DEBUG_HAS_OPENCV=1)
target_include_directories(rgb_target PUBLIC include PRIVATE ${Boost_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(rgb_target PUBLIC ${OpenCV_LIBS})
find_package(Threads REQUIRED)
add_library(visual_pick STATIC src/visual_pick.cpp)
target_include_directories(visual_pick PUBLIC include PRIVATE ${Boost_INCLUDE_DIRS})
target_link_libraries(visual_pick PUBLIC Threads::Threads)
add_executable(bridge_node src/bridge_node.cpp src/controller_registry.cpp)
target_link_libraries(bridge_node visual_pick rgb_target)
target_link_libraries(bridge_node object_detection)
target_include_directories(bridge_node PRIVATE include)
ament_target_dependencies(bridge_node
  ament_index_cpp rclcpp rclcpp_action std_msgs sensor_msgs geometry_msgs nav_msgs diagnostic_msgs
  trajectory_msgs shape_msgs tf2 tf2_ros microros_stm32h723_bridge rosidl_typesupport_introspection_cpp)
target_link_libraries(bridge_node Boost::system Boost::thread ZLIB::ZLIB)
if(controller_manager_msgs_FOUND)
  target_compile_definitions(bridge_node PRIVATE EASY_DEBUG_HAS_CONTROLLER_MANAGER=1)
  ament_target_dependencies(bridge_node controller_manager_msgs)
endif()
if(moveit_msgs_FOUND AND control_msgs_FOUND AND controller_manager_msgs_FOUND)
  target_sources(bridge_node PRIVATE src/position_motion.cpp)
  ament_target_dependencies(bridge_node moveit_msgs control_msgs controller_manager_msgs)
else()
  target_sources(bridge_node PRIVATE src/position_motion_unavailable.cpp)
  message(WARNING "MoveIt2 position execution disabled: optional ROS message packages unavailable")
endif()
